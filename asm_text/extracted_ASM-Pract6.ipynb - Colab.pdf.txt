from sklearn.datasets import load_diabetes
import pandas as pd
import statsmodels.api as sm
diabetes = load_diabetes(as_frame=True)
df = diabetes.frame
print(df.head())
# Correlation
print(df[["bmi", "target"]].corr())
# Simple linear regression
X = sm.add_constant(df["bmi"])
y = df["target"]
model = sm.OLS(y, X).fit()
print(model.summary())
