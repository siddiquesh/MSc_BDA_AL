from scipy import stats

n= len(delivery_time)

#sample mean and standard deviation
s_mean = np.mean(delivery_time)
s_std = np.std(delivery_time,ddof=1)

#degree of freedom
df= n-1

#standard error
SE= s_std / np.sqrt(n)

#95% confidence interval
t_critical = stats.t.ppf(0.975,df)
margin_error = t_critical * SE

lower= s_mean - margin_error
upper = s_mean + margin_error

print("Sample Mean = ",s_mean)
print("Sample SD = ",s_std)
print("Degrees of freedom = ",df)
print("Standard Error = ",SE)
print("Margin of error = ",margin_error)
print("95% Confidence Interval = ",(lower,upper))

#Byesian Updating for medical Testing

#prior probability
prior_dieases = 0.01

#test characteristics
sensitivity = 0.95
specificity = 0.90

#probability of positive test among people without disease
false_positive_rate = 1 - specificity

#Bayes's theorem fter first positive test
prob_positive = ( sensitivity * prior_dieases ) + ( false_positive_rate * (1 - prior_dieases))

posterior_1 =( sensitivity * prior_dieases ) / prob_positive

print("probability of positive test: ",prob_positive)
print("posterior proability after first positive test:",posterior_1)
